  <behavior>
All four endpoints must satisfy their exact contracts from FRD §API Endpoints and the user's locked error response shape decision.

Test cases (input → expected output):

**GET /api/tasks:**
- Empty DB → 200 `[]`
- Two tasks inserted → 200 array with both tasks, ordered by created_at ASC
- Each task object has `id` (number), `title` (string), `completed` (boolean), `created_at` (string)

**POST /api/tasks:**
- `{ title: "Buy groceries" }` → 201 `{ id: 1, title: "Buy groceries", completed: false, created_at: "..." }`
- `{ title: "  Buy groceries  " }` (whitespace padded) → 201 with trimmed title
- `{ title: "" }` → 400 `{ error: "Title is required" }`
- `{ title: "   " }` (whitespace only) → 400 `{ error: "Title is required" }`
- `{ title: "x".repeat(501) }` → 400 `{ error: "Title must be 500 characters or fewer" }`
- `{ title: "x".repeat(500) }` → 201 (exact boundary)
- Missing title field → 400 `{ error: "Title is required" }`

**PATCH /api/tasks/:id:**
- Valid id, `{ completed: true }` → 200 updated task with `completed: true`
- Valid id, `{ completed: false }` (toggle back) → 200 with `completed: false`
- Unknown id 9999, `{ completed: true }` → 404 `{ error: "Task not found" }`
- Valid id, `{ completed: "yes" }` → 400 `{ error: "completed must be a boolean" }`
- Valid id, `{ completed: 1 }` (integer, not boolean) → 400 flat error

**DELETE /api/tasks/:id:**
- Valid id → 204 empty body
- Unknown id 9999 → 404 `{ error: "Task not found" }`
- After DELETE: GET /api/tasks no longer includes the deleted task

**Error shape validation (for ALL error cases):**
- Response body has `error` property of type `string` (NOT an object)
- `typeof res.body.error === 'string'` must be true
  </behavior>
  <implementation>
Use supertest to make HTTP requests against the Express `app` (not a bound port). Use an in-memory SQLite database by setting `DATABASE_URL=:memory:` in the test environment. Each `describe` block clears the tasks table in `beforeEach` for isolation.

**Key implementation pattern:**
```javascript
// Set in-memory DB before requiring server
process.env.DATABASE_URL = ":memory:";
const { app, server } = require("../server.js");
const request = require("supertest");

afterAll(() => server.close());
```

Note: `better-sqlite3` supports `:memory:` as the database path for an in-memory database. This is a valid path for testing.

**Test structure:**
```javascript
describe("GET /api/tasks", () => { ... });
describe("POST /api/tasks", () => { ... });
describe("PATCH /api/tasks/:id", () => { ... });
describe("DELETE /api/tasks/:id", () => { ... });
```

**Isolation strategy:** In `beforeEach` for each describe block, delete all rows: `db.prepare("DELETE FROM tasks").run()`. Import `db` from `../db.js` to access the connection directly for setup/teardown.
  </implementation>
</feature>

```javascript
"use strict";

// Use in-memory SQLite so tests don't touch data/todo.db
// Must be set before requiring server.js (which requires db.js on load)
process.env.DATABASE_URL = ":memory:";
process.env.PORT = "0"; // Let OS assign a free port

const request = require("supertest");
const { app, server } = require("../server.js");
const db = require("../db.js");

afterAll((done) => {
  server.close(done);
});

// Clear all tasks before each test for isolation
beforeEach(() => {
  db.prepare("DELETE FROM tasks").run();
});

// ============================================================
// GET /api/tasks
// ============================================================
describe("GET /api/tasks", () => {
  it("returns 200 with an empty array when no tasks exist", async () => {
    const res = await request(app).get("/api/tasks");
    expect(res.status).toBe(200);
    expect(Array.isArray(res.body)).toBe(true);
    expect(res.body).toHaveLength(0);
  });

  it("returns all tasks ordered by created_at ASC", async () => {
    // Insert two tasks
    db.prepare("INSERT INTO tasks (title) VALUES (?)").run("First task");
    db.prepare("INSERT INTO tasks (title) VALUES (?)").run("Second task");

    const res = await request(app).get("/api/tasks");
    expect(res.status).toBe(200);
    expect(res.body).toHaveLength(2);
    expect(res.body[0].title).toBe("First task");
    expect(res.body[1].title).toBe("Second task");
  });

  it("returns task objects with correct shape (id, title, completed, created_at)", async () => {
    db.prepare("INSERT INTO tasks (title) VALUES (?)").run("Shape test");

    const res = await request(app).get("/api/tasks");
    expect(res.status).toBe(200);
    const task = res.body[0];
    expect(typeof task.id).toBe("number");
    expect(typeof task.title).toBe("string");
    expect(typeof task.completed).toBe("boolean");
    expect(typeof task.created_at).toBe("string");
  });

  it("returns completed as boolean false by default", async () => {
    db.prepare("INSERT INTO tasks (title) VALUES (?)").run("Completion check");

    const res = await request(app).get("/api/tasks");
    expect(res.body[0].completed).toBe(false);
  });
});

// ============================================================
// POST /api/tasks
// ============================================================
describe("POST /api/tasks", () => {
  it("creates a task and returns 201 with the created object", async () => {
    const res = await request(app)
      .post("/api/tasks")
      .send({ title: "Buy groceries" });

    expect(res.status).toBe(201);
    expect(typeof res.body.id).toBe("number");
    expect(res.body.title).toBe("Buy groceries");
    expect(res.body.completed).toBe(false);
    expect(typeof res.body.created_at).toBe("string");
  });

  it("trims whitespace from the title before saving", async () => {
    const res = await request(app)
      .post("/api/tasks")
      .send({ title: "  Buy groceries  " });

    expect(res.status).toBe(201);
    expect(res.body.title).toBe("Buy groceries");
  });

  it("returns 400 when title is empty string", async () => {
    const res = await request(app)
      .post("/api/tasks")
      .send({ title: "" });

    expect(res.status).toBe(400);
    expect(typeof res.body.error).toBe("string"); // flat format
    expect(res.body.error).toBe("Title is required");
  });

  it("returns 400 when title is whitespace only", async () => {
    const res = await request(app)
      .post("/api/tasks")
      .send({ title: "   " });

    expect(res.status).toBe(400);
    expect(typeof res.body.error).toBe("string");
    expect(res.body.error).toBe("Title is required");
  });

  it("returns 400 when title field is missing", async () => {
    const res = await request(app)
      .post("/api/tasks")
      .send({});

    expect(res.status).toBe(400);
    expect(typeof res.body.error).toBe("string");
    expect(res.body.error).toBe("Title is required");
  });

  it("returns 400 when title exceeds 500 characters", async () => {
    const res = await request(app)
      .post("/api/tasks")
      .send({ title: "x".repeat(501) });

    expect(res.status).toBe(400);
    expect(typeof res.body.error).toBe("string");
    expect(res.body.error).toBe("Title must be 500 characters or fewer");
  });

  it("accepts a title of exactly 500 characters", async () => {
    const res = await request(app)
      .post("/api/tasks")
      .send({ title: "x".repeat(500) });

    expect(res.status).toBe(201);
    expect(res.body.title).toHaveLength(500);
  });
});

// ============================================================
// PATCH /api/tasks/:id
// ============================================================
describe("PATCH /api/tasks/:id", () => {
  let taskId;

  beforeEach(() => {
    const result = db
      .prepare("INSERT INTO tasks (title) VALUES (?)")
      .run("Test task");
    taskId = result.lastInsertRowid;
  });

  it("updates completed to true and returns 200 with updated task", async () => {
    const res = await request(app)
      .patch(`/api/tasks/${taskId}`)
      .send({ completed: true });

    expect(res.status).toBe(200);
    expect(res.body.id).toBe(taskId);
    expect(res.body.completed).toBe(true);
    expect(res.body.title).toBe("Test task");
  });

  it("toggles completed back to false", async () => {
    // Mark complete first
    db.prepare("UPDATE tasks SET completed = 1 WHERE id = ?").run(taskId);

    const res = await request(app)
      .patch(`/api/tasks/${taskId}`)
      .send({ completed: false });

    expect(res.status).toBe(200);
    expect(res.body.completed).toBe(false);
  });

  it("returns 404 when task ID does not exist", async () => {
    const res = await request(app)
      .patch("/api/tasks/9999")
      .send({ completed: true });

    expect(res.status).toBe(404);
    expect(typeof res.body.error).toBe("string"); // flat format
    expect(res.body.error).toBe("Task not found");
  });

  it("returns 400 when completed is a string", async () => {
    const res = await request(app)
      .patch(`/api/tasks/${taskId}`)
      .send({ completed: "yes" });

    expect(res.status).toBe(400);
    expect(typeof res.body.error).toBe("string");
    expect(res.body.error).toBe("completed must be a boolean");
  });

  it("returns 400 when completed is an integer (not a boolean)", async () => {
    const res = await request(app)
      .patch(`/api/tasks/${taskId}`)
      .send({ completed: 1 });

    expect(res.status).toBe(400);
    expect(typeof res.body.error).toBe("string");
    expect(res.body.error).toBe("completed must be a boolean");
  });

  it("returns 400 when completed is missing from body", async () => {
    const res = await request(app)
      .patch(`/api/tasks/${taskId}`)
      .send({});

    expect(res.status).toBe(400);
    expect(typeof res.body.error).toBe("string");
    expect(res.body.error).toBe("completed must be a boolean");
  });
});

// ============================================================
// DELETE /api/tasks/:id
// ============================================================
describe("DELETE /api/tasks/:id", () => {
  let taskId;

  beforeEach(() => {
    const result = db
      .prepare("INSERT INTO tasks (title) VALUES (?)")
      .run("Task to delete");
    taskId = result.lastInsertRowid;
  });

  it("deletes the task and returns 204 with no body", async () => {
    const res = await request(app).delete(`/api/tasks/${taskId}`);

    expect(res.status).toBe(204);
    expect(res.body).toEqual({});
  });

  it("task is removed from GET /api/tasks after deletion", async () => {
    await request(app).delete(`/api/tasks/${taskId}`);

    const res = await request(app).get("/api/tasks");
    expect(res.status).toBe(200);
    const ids = res.body.map((t) => t.id);
    expect(ids).not.toContain(taskId);
  });

  it("returns 404 when task ID does not exist", async () => {
    const res = await request(app).delete("/api/tasks/9999");

    expect(res.status).toBe(404);
    expect(typeof res.body.error).toBe("string"); // flat format
    expect(res.body.error).toBe("Task not found");
  });
});

// ============================================================
// Error shape: ALL errors must use flat { error: string } format
// (Not nested { error: { code, message } } — user locked decision)
// ============================================================
describe("Error response shape — flat format enforcement", () => {
  it("400 error has string error property, not an object", async () => {
    const res = await request(app)
      .post("/api/tasks")
      .send({ title: "" });

    expect(typeof res.body.error).toBe("string");
    expect(res.body.error).not.toBeNull();
    // Confirm it is NOT the nested shape
    expect(typeof res.body.error).not.toBe("object");
  });

  it("404 error has string error property, not an object", async () => {
    const res = await request(app)
      .patch("/api/tasks/9999")
      .send({ completed: true });

    expect(typeof res.body.error).toBe("string");
    expect(typeof res.body.error).not.toBe("object");
  });
});

// ============================================================
// CORS — Access-Control-Allow-Origin: * on all responses
// ============================================================
describe("CORS headers", () => {
  it("GET /api/tasks includes Access-Control-Allow-Origin: *", async () => {
    const res = await request(app).get("/api/tasks");
    expect(res.headers["access-control-allow-origin"]).toBe("*");
  });

  it("POST /api/tasks includes Access-Control-Allow-Origin: *", async () => {
    const res = await request(app)
      .post("/api/tasks")
      .send({ title: "CORS test" });
    expect(res.headers["access-control-allow-origin"]).toBe("*");
  });
});
```
