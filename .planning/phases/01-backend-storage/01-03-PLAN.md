**Error response shape — USER LOCKED DECISION (flat format):**
ALL errors use `{ "error": "message string" }` — flat, no nested object. This overrides the FRD/TechArch which specifies a nested `{ error: { code, message } }` shape. Use flat format ONLY.

**SQL from TechArch §4.4 — copy exactly:**

```sql
-- GET all tasks
SELECT id, title, completed, created_at FROM tasks ORDER BY created_at ASC;

-- POST create task
INSERT INTO tasks (title) VALUES (?);
SELECT id, title, completed, created_at FROM tasks WHERE id = last_insert_rowid();

-- PATCH update task
UPDATE tasks SET completed = ? WHERE id = ?;
SELECT id, title, completed, created_at FROM tasks WHERE id = ?;

-- DELETE task
DELETE FROM tasks WHERE id = ?;
```

**Endpoint specs from FRD §API Endpoints (exact):**
- `GET /api/tasks` → 200 array of task objects, ordered by created_at ASC
- `POST /api/tasks` → body: `{ title: string }`, 201 with created task object
- `PATCH /api/tasks/:id` → body: `{ completed: boolean }`, 200 with updated task object
- `DELETE /api/tasks/:id` → 204 No Content

**Validation rules (from FRD §F0 and TechArch §4.4):**
- `POST`: title must be present, non-empty, non-whitespace-only → 400 `{ "error": "Title is required" }`
- `POST`: title must be ≤ 500 chars → 400 `{ "error": "Title must be 500 characters or fewer" }`
- `PATCH`: `completed` must be a boolean (`typeof completed === 'boolean'`) → 400 `{ "error": "completed must be a boolean" }`
- `PATCH`/`DELETE`: if task ID not found → 404 `{ "error": "Task not found" }`
- Any unhandled exception → 500 `{ "error": "Internal server error" }` (never leak stack trace or DB error)
- All server errors logged to `console.error` before sending safe response

**Complete implementation:**

```javascript
"use strict";

const db = require("../db.js");

/**
 * GET /api/tasks
 * Returns all tasks ordered by created_at ASC.
 */
function getAllTasks(req, res) {
  try {
    const tasks = db
      .prepare(
        "SELECT id, title, completed, created_at FROM tasks ORDER BY created_at ASC"
      )
      .all();
    // better-sqlite3 stores BOOLEAN as 0/1 — normalize to JS boolean
    const normalized = tasks.map((t) => ({
      ...t,
      completed: t.completed === 1 || t.completed === true,
    }));
    return res.status(200).json(normalized);
  } catch (err) {
    console.error("GET /api/tasks error:", err);
    return res.status(500).json({ error: "Internal server error" });
  }
}

/**
 * POST /api/tasks
 * Creates a new task. Body: { title: string }
 */
function createTask(req, res) {
  try {
    const { title } = req.body;

    // Validate title
    if (typeof title !== "string" || title.trim().length === 0) {
      return res.status(400).json({ error: "Title is required" });
    }
    if (title.trim().length > 500) {
      return res
        .status(400)
        .json({ error: "Title must be 500 characters or fewer" });
    }

    const trimmedTitle = title.trim();

    // Insert then fetch the created row
    db.prepare("INSERT INTO tasks (title) VALUES (?)").run(trimmedTitle);
    const task = db
      .prepare(
        "SELECT id, title, completed, created_at FROM tasks WHERE id = last_insert_rowid()"
      )
      .get();

    const result = {
      ...task,
      completed: task.completed === 1 || task.completed === true,
    };

    return res.status(201).json(result);
  } catch (err) {
    console.error("POST /api/tasks error:", err);
    return res.status(500).json({ error: "Internal server error" });
  }
}

/**
 * PATCH /api/tasks/:id
 * Updates the completed state of a task. Body: { completed: boolean }
 */
function updateTask(req, res) {
  try {
    const id = parseInt(req.params.id, 10);
    const { completed } = req.body;

    // Validate completed is a boolean
    if (typeof completed !== "boolean") {
      return res
        .status(400)
        .json({ error: "completed must be a boolean" });
    }

    // Check task exists
    const existing = db
      .prepare("SELECT id FROM tasks WHERE id = ?")
      .get(id);
    if (!existing) {
      return res.status(404).json({ error: "Task not found" });
    }

    // Update then fetch the updated row
    db.prepare("UPDATE tasks SET completed = ? WHERE id = ?").run(
      completed ? 1 : 0,
      id
    );
    const task = db
      .prepare(
        "SELECT id, title, completed, created_at FROM tasks WHERE id = ?"
      )
      .get(id);

    const result = {
      ...task,
      completed: task.completed === 1 || task.completed === true,
    };

    return res.status(200).json(result);
  } catch (err) {
    console.error("PATCH /api/tasks/:id error:", err);
    return res.status(500).json({ error: "Internal server error" });
  }
}

/**
 * DELETE /api/tasks/:id
 * Permanently deletes a task. Returns 204 No Content.
 */
function deleteTask(req, res) {
  try {
    const id = parseInt(req.params.id, 10);

    // Check task exists
    const existing = db
      .prepare("SELECT id FROM tasks WHERE id = ?")
      .get(id);
    if (!existing) {
      return res.status(404).json({ error: "Task not found" });
    }

    db.prepare("DELETE FROM tasks WHERE id = ?").run(id);

    return res.status(204).send();
  } catch (err) {
    console.error("DELETE /api/tasks/:id error:", err);
    return res.status(500).json({ error: "Internal server error" });
  }
}

module.exports = { getAllTasks, createTask, updateTask, deleteTask };
```

**Key implementation notes:**
- `better-sqlite3` stores SQLite BOOLEAN as 0/1 integers — always normalize to JS `true`/`false` before sending responses
- Use `parseInt(req.params.id, 10)` for route params — Express delivers them as strings
- `last_insert_rowid()` is a SQLite function available in the same connection after an INSERT
- Trim titles before INSERT to match validation behavior
- `console.error` on every caught exception before the safe 500 response

### `server.js`

Entry point implementing the TechArch Appendix A startup sequence exactly:

```javascript
"use strict";

const path = require("path");
require("dotenv").config();

// Step 1: Initialize database (runs CREATE TABLE IF NOT EXISTS on require)
// Fatal exit if DB cannot be opened or schema cannot be created (handled in db.js)
const db = require("./db.js");

// Step 2: Configure Express
const express = require("express");
const helmet = require("helmet");
const cors = require("cors");
const taskRoutes = require("./routes/tasks.js");

const app = express();
const PORT = process.env.PORT || 3000;

// Step 3: Apply middleware
// helmet — security headers (X-Content-Type-Options, X-Frame-Options, CSP, etc.)
app.use(helmet());

// cors — Allow all origins per user decision (Access-Control-Allow-Origin: *)
app.use(cors());

// JSON body parser
app.use(express.json());

// Static files from public/ (Phase 2 frontend)
app.use(express.static(path.join(__dirname, "public")));

// Step 4: Mount API routes
app.use("/api/tasks", taskRoutes);

// Step 5: Global error handler (catch-all for unhandled errors)
// Must have 4 parameters for Express to recognize it as error middleware
// eslint-disable-next-line no-unused-vars
app.use((err, req, res, next) => {
  console.error("Unhandled error:", err);
  res.status(500).json({ error: "Internal server error" });
});

// Step 6: Start the HTTP server
const server = app.listen(PORT, () => {
  console.log(`TodoApp listening on http://localhost:${PORT}`);
});

module.exports = { app, server };
```

**Implementation notes:**
- `server.js` exports `{ app, server }` so supertest can import `app` without binding to a port during tests
- `db` is required at the top — if db.js calls `process.exit(1)`, the server never reaches `app.listen`
- `cors()` with no arguments uses default which sets `Access-Control-Allow-Origin: *` — this satisfies the user's locked CORS decision
- `helmet()` with no arguments applies all default security headers including `X-Content-Type-Options: nosniff`, `X-Frame-Options: DENY`, and `Content-Security-Policy`
- The global error handler (4-param middleware) catches any error thrown by route handlers that wasn't caught internally
- `module.exports = { app, server }` — `app` is used by supertest in tests; `server` is returned so tests can call `server.close()` for clean teardown
